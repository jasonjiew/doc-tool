def paragraph_text(paragraph) -> str:
    """段落可见文本：跳过域指令，保留域缓存值（Word 刷新前的可读结果）。"""
    parts: List[str] = []
    for node in paragraph.iter():
        if node.tag == qn("fldChar"):
            kind = node.get(qn("fldCharType"))
            if kind == "begin":
                in_instruction += 1
            elif kind == "separate" and in_instruction:
                in_instruction = 0
            elif kind == "end" and not in_instruction:
                in_instruction = 0
            continue
        if in_instruction:
            continue
        if node.tag == qn("t"):
            parts.append(node.text or "")
        elif node.tag in (qn("br"), qn("cr")):
            parts.append("\n")
        elif node.tag == qn("tab"):
            parts.append("\t")
    return "".join(parts).strip()
